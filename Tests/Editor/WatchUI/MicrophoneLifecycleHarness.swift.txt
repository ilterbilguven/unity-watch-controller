@objc private enum WCSessionActivationState: Int { case notActivated, inactive, activated }
@objc private protocol WCSessionDelegate: AnyObject {
    @objc optional func session(_ session: WCSession, didFinish fileTransfer: WCSessionFileTransfer, error: Error?)
    @objc optional func session(_ session: WCSession, activationDidCompleteWith activationState: WCSessionActivationState, error: Error?)
}
private final class WCSessionFile: NSObject {
    let fileURL: URL
    let metadata: [String: Any]?
    init(url: URL, metadata: [String: Any]?) {
        fileURL = url
        self.metadata = metadata
    }
}
private final class WCSessionFileTransfer: NSObject {
    let file: WCSessionFile
    init(file: WCSessionFile) { self.file = file }
}
private final class WCSession: NSObject {
    static let `default` = WCSession()
    weak var delegate: WCSessionDelegate?
    var activationState = WCSessionActivationState.activated
    var isReachable = true
    var transfers: [WCSessionFileTransfer] = []
    func activate() {}
    func sendMessage(_ message: [String: Any], replyHandler: (([String: Any]) -> Void)?, errorHandler: ((Error) -> Void)?) {}
    @discardableResult func transferFile(_ url: URL, metadata: [String: Any]?) -> WCSessionFileTransfer {
        let transfer = WCSessionFileTransfer(file: WCSessionFile(url: url, metadata: metadata))
        transfers.append(transfer)
        return transfer
    }
    func complete(_ index: Int, error: Error? = nil) {
        delegate?.session?(self, didFinish: transfers[index], error: error)
    }
    func finishActivation() {
        activationState = .activated
        delegate?.session?(self, activationDidCompleteWith: .activated, error: nil)
    }
}
